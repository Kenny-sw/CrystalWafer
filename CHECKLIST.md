## 🔗 Интеграция с WaferController

### WaferController.cs
Добавить поддержку WaferTemplate:

```csharp
public void LoadFromTemplate(WaferTemplate template)
{
    var parameters = new WaferMapParameters
    {
        DiameterMm = template.DiameterMm,
        CrystalWidthMm = template.CrystalWidthMm,
        CrystalHeightMm = template.CrystalHeightMm,
        StreetMm = template.StreetMm,
        OffsetXMm = template.OffsetXMm,
        OffsetYMm = template.OffsetYMm,
        MirrorX = template.MirrorX,
        MirrorY = template.MirrorY,
        SwapOrientation = false
    };
    
    BeginMapCreation(parameters);
}

public WaferTemplate ExportToTemplate(string templateName)
{
    return new WaferTemplate
    {
        TemplateName = templateName,
        DiameterMm = this.WaferDiameter,
        CrystalWidthMm = this.CrystalWidthRaw / 1000f,
        CrystalHeightMm = this.CrystalHeightRaw / 1000f,
        StreetMm = this.StreetMm,
        OffsetXMm = this.OffsetXMm,
        OffsetYMm = this.OffsetYMm,
        MirrorX = this.MirrorX,
        MirrorY = this.MirrorY,
        EdgeExclusionMm = this.EdgeExclusionMm
    };
}
```
